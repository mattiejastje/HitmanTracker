struct TimeManager {
  int32_t time;  // offset 0x48
};

struct Suits {
  int32_t current_suit;  // offset 0x0
  int32_t starting_suit;  // offset 0x4
};

struct SuitContainer {
  Suits suits;  // offset 0xfd0
};

struct Settings {
  SuitContainer* suit_container;  // offset 0xa40
  int32_t difficulty;  // offset 0x6664
};

struct Game {
  TimeManager* time_manager;  // offset 0x41f820
  Settings* settings;  // offset 0x41f83c
  std::array<int32_t, 0x42> stats;  // offset 0x5b2538
};
