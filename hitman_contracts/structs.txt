struct PropertyType {
  uint32_t fourcc;  // offset 0x0
  int32_t fourcc_length;  // offset 0x4
  int32_t index;  // offset 0x8
  int32_t size;  // offset 0xc
  int32_t type_flags;  // offset 0x10
};

struct Property {
  int32_t key_length;  // offset 0x0
  PropertyType* type;  // offset 0x4
  int32_t size;  // offset 0x8
  char[64] key;  // offset 0xc
};

struct PropertyManagerRecord {
  int32_t record_size;  // offset 0x0
  int8_t is_active;  // offset 0x4
  Property property;  // offset 0x5
};

struct SmallString {
  char* text;  // offset 0x0
};

};

struct Engine {

};

struct PropertyManager {

  int32_t data_capacity;  // offset 0x4

  int32_t data_used;  // offset 0xc
};

};

  int32_t innocents_killed;  // offset 0xb27
